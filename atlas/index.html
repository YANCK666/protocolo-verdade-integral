function drawLight(date) { const W = RES, H = RES; const shd = sctx.createImageData(W, H); // ===== SOL (ponteiro do relógio) ===== const jd = date.getTime() / 86400000 + 2440587.5; const n = jd - 2451545.0; // Longitude eclíptica do Sol const L = ((280.460 + 0.9856474 * n) % 360 + 360) % 360; const g = (((357.528 + 0.9856003 * n) % 360) + 360) % 360 * Math.PI / 180; const lambda = (L + 1.915 * Math.sin(g) + 0.020 * Math.sin(2 * g)) * Math.PI / 180; const eps = (23.439 - 0.0000004 * n) * Math.PI / 180; // Ascensão reta e declinação reais let sunRa = Math.atan2(Math.cos(eps) * Math.sin(lambda), Math.cos(lambda)) * 180 / Math.PI; sunRa = ((sunRa % 360) + 360) % 360; const sunDec = Math.asin(Math.sin(eps) * Math.sin(lambda)) * 180 / Math.PI; const gmst = gmstDeg(date); const sunAngle = (sunRa - gmst) * Math.PI / 180; // ângulo do Sol no mapa // ===== SOMBRA sempre do lado oposto ao Sol ===== for (let y = 0; y < H; y++) { for (let x = 0; x < W; x++) { const px = x * (SIZE / W); const py = y * (SIZE / H); const dx = px - CX, dy = py - CY; const dist = Math.sqrt(dx * dx + dy * dy); const idx = (y * W + x) * 4; if (dist > R) continue; const ptAngle = Math.atan2(dx, -dy); let diff = ptAngle - sunAngle; while (diff > Math.PI) diff -= 2 * Math.PI; while (diff < -Math.PI) diff += 2 * Math.PI; // Intensidade da sombra (máxima no lado oposto ao Sol) const shadowIntensity = (1 - Math.cos(diff)) / 2; const radialFade = Math.min(1, dist / R); shd.data[idx] = 3; shd.data[idx + 1] = 6; shd.data[idx + 2] = 22; shd.data[idx + 3] = Math.floor(shadowIntensity * radialFade * 230); } } sctx.putImageData(shd, 0, 0); solarShadow.setAttribute("href", shadowCanvas.toDataURL("image/png")); // ===== Posiciona o Sol (ponteiro) ===== const sunRho = R * (90 - sunDec) / 180; const sunX = CX + sunRho * Math.sin(sunAngle); const sunY = CY - sunRho * Math.cos(sunAngle); sunIcon.setAttribute("cx", sunX); sunIcon.setAttribute("cy", sunY); sunIcon.style.display = (sunRho <= R + 15) ? "" : "none"; // ===== Lua (segue o mesmo referencial do Sol) ===== try { const illum = SunCalc.getMoonIllumination(date); const elong = illum.phase * 2 * Math.PI; // Posição aproximada mas no mesmo sistema do Sol const moonRa = (sunRa + (elong * 180 / Math.PI) + 360) % 360; const moonDec = 5.1 * Math.sin(elong); // inclinação aproximada da órbita lunar const moonRho = R * (90 - moonDec) / 180; const moonAngle = (moonRa - gmst) * Math.PI / 180; const moonX = CX + moonRho * Math.sin(moonAngle); const moonY = CY - moonRho * Math.cos(moonAngle); if (moonRho <= R) { moonIcon.setAttribute("cx", moonX); moonIcon.setAttribute("cy", moonY); moonClipCircle.setAttribute("cx", moonX); moonClipCircle.setAttribute("cy", moonY); // Fase da Lua const angleToSun = Math.atan2(sunY - moonY, sunX - moonX); const phase = illum.fraction; const shadowDistance = (1 - phase) * 24; moonShadow.setAttribute("cx", moonX - Math.cos(angleToSun) * shadowDistance); moonShadow.setAttribute("cy", moonY - Math.sin(angleToSun) * shadowDistance); moonShadow.setAttribute("r", 12); } } catch (e) {} }
