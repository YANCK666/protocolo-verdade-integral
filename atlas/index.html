/* ===================================================== MAPA LUCIFERIANO — ILUMINAÇÃO SOLAR E SOMBRA Projeção azimutal equidistante: centro = Polo Norte borda = Polo Sul ===================================================== */ (function () { const svg = document.getElementById("map"); const worldGroup = document.getElementById("worldGroup"); if (!svg || !worldGroup) { console.error("Mapa ou worldGroup não encontrado."); return; } // Geometria do mapa (viewBox 800 x 800) const CX = 400; const CY = 400; const RADIUS = 380; // Resolução da imagem de sombra const SIZE = 800; // Criar imagem da sombra dentro do SVG let shadowImage = document.getElementById("solarShadowImage"); if (!shadowImage) { shadowImage = document.createElementNS( "http://www.w3.org/2000/svg", "image" ); shadowImage.setAttribute("id", "solarShadowImage"); shadowImage.setAttribute("x", "0"); shadowImage.setAttribute("y", "0"); shadowImage.setAttribute("width", SIZE); shadowImage.setAttribute("height", SIZE); shadowImage.setAttribute("preserveAspectRatio", "none"); // Coloca a sombra acima do território // e antes das linhas e elementos fixos. svg.insertBefore(shadowImage, svg.querySelector("#staticGroup")); } // Criar imagem em memória para desenhar a noite const canvas = document.createElement("canvas"); canvas.width = SIZE; canvas.height = SIZE; const ctx = canvas.getContext("2d"); const pixels = ctx.createImageData(SIZE, SIZE); // Declinação solar aproximada function solarDeclination(date) { const start = new Date(Date.UTC(date.getUTCFullYear(), 0, 0)); const day = Math.floor((date - start) / 86400000); const gamma = 2 * Math.PI / 365 * (day - 1); return ( 0.006918 - 0.399912 * Math.cos(gamma) + 0.070257 * Math.sin(gamma) - 0.006758 * Math.cos(2 * gamma) + 0.000907 * Math.sin(2 * gamma) - 0.002697 * Math.cos(3 * gamma) + 0.00148 * Math.sin(3 * gamma) ); } // Longitude aproximada do ponto onde o Sol está a pino function solarLongitude(date) { const hours = date.getUTCHours() + date.getUTCMinutes() / 60 + date.getUTCSeconds() / 3600; return (180 - hours * 15 + 360) % 360; } function drawSolarShadow() { const now = new Date(); const declination = solarDeclination(now); const sunLongitude = solarLongitude(now); const sinDeclination = Math.sin(declination); const cosDeclination = Math.cos(declination); for (let y = 0; y < SIZE; y++) { for (let x = 0; x < SIZE; x++) { const dx = x - CX; const dy = y - CY; const radius = Math.sqrt(dx * dx + dy * dy); const index = (y * SIZE + x) * 4; // Fora do disco: transparente if (radius > RADIUS) { pixels.data[index + 3] = 0; continue; } // Converter posição do disco em latitude const latitude = Math.PI / 2 - (radius / RADIUS) * Math.PI; // Longitude na projeção polar const longitude = Math.atan2(dx, -dy) * 180 / Math.PI; const hourAngle = (longitude - sunLongitude) * Math.PI / 180; // Altura solar: positiva = dia; negativa = noite const solarHeight = Math.sin(latitude) * sinDeclination + Math.cos(latitude) * cosDeclination * Math.cos(hourAngle); if (solarHeight < 0) { // Sombra: azul escuro translúcido pixels.data[index] = 3; pixels.data[index + 1] = 8; pixels.data[index + 2] = 35; pixels.data[index + 3] = 205; } else { // Dia: transparente, preserva a luz e o mapa pixels.data[index + 3] = 0; } } } ctx.putImageData(pixels, 0, 0); shadowImage.setAttribute( "href", canvas.toDataURL("image/png") ); } // Atualização inicial e periódica drawSolarShadow(); setInterval(drawSolarShadow, 60000); // Permite atualizar a sombra manualmente window.atualizarSombraSolar = drawSolarShadow; })();
